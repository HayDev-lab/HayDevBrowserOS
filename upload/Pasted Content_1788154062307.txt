<script type="module">
import * as THREE from 'three';

// ============================================================
// EMOTION TEXTURES (base64 placeholders — в реальности загружаются из файлов)
// ============================================================
const EMOTION_TEXTURES = {
  neutral: null,
  happy: null,
  smile: null,
  excited: null,
  sad: null,
  angry: null,
  surprised: null,
  thinking: null,
  confused: null,
  focused: null,
  speaking: null,
  listening: null,
  blink: null
};

// В реальном проекте здесь будут пути к файлам:
// '/avatar/textures/emotions/neutral.webp' и т.д.

const EMOTION_NAMES = Object.keys(EMOTION_TEXTURES);

// ============================================================
// EASING
// ============================================================
function easeInOutCubic(t) {
  return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
}

// ============================================================
// MAIN AVATAR CLASS
// ============================================================
class EmotionAvatar {
  constructor(canvas) {
    this.canvas = canvas;
    this.currentEmotion = 'neutral';
    this.targetEmotion = 'neutral';
    this.transitionProgress = 1;
    this.transitionDuration = 0.35;
    this.transitionStart = 0;

    this.blinkEnabled = true;
    this.eyeMovementEnabled = true;
    this.microEnabled = true;
    this.speaking = false;
    this.speechLevel = 0;
    this.useTextures = true;

    this._tempVec = new THREE.Vector3();

    this.init();
  }

  init() {
    // Renderer
    this.renderer = new THREE.WebGLRenderer({
      canvas: this.canvas,
      antialias: true,
      alpha: true
    });
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.setSize(this.canvas.parentElement.clientWidth, this.canvas.parentElement.clientHeight);
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
    this.renderer.toneMappingExposure = 1.2;

    // Scene
    this.scene = new THREE.Scene();

    // Camera
    this.camera = new THREE.PerspectiveCamera(30, 1, 0.1, 100);
    this.camera.position.set(0, 0, 4);

    // Lights
    const ambientLight = new THREE.AmbientLight(0x4080ff, 0.6);
    this.scene.add(ambientLight);

    const keyLight = new THREE.DirectionalLight(0x80d0ff, 1.5);
    keyLight.position.set(2, 3, 4);
    this.scene.add(keyLight);

    const fillLight = new THREE.DirectionalLight(0x4080ff, 0.5);
    fillLight.position.set(-2, 1, 2);
    this.scene.add(fillLight);

    const rimLight = new THREE.DirectionalLight(0x00d4ff, 0.4);
    rimLight.position.set(0, 2, -3);
    this.scene.add(rimLight);

    // Create avatar plane (will display textures)
    const planeGeo = new THREE.PlaneGeometry(2.5, 3.2);
    const planeMat = new THREE.MeshBasicMaterial({
      transparent: true,
      opacity: 1
    });

    this.avatarMesh = new THREE.Mesh(planeGeo, planeMat);
    this.scene.add(this.avatarMesh);

    // Load initial texture
    this.loadTexture('neutral');

    // Blink state
    this.blinkState = { phase: 'open', timer: 0, nextBlink: this.randomBlinkInterval(), duration: 0, blinkOpacity: 1 };

    // Eye movement (simulated via texture offset)
    this.eyeState = { targetX: 0, targetY: 0, currentX: 0, currentY: 0, timer: 0, nextMove: 2 };

    // Micro expression
    this.microState = { timer: 0, nextMicro: 5, intensity: 0 };

    // FPS tracking
    this.frameCount = 0;
    this.fpsTime = 0;
    this.currentFPS = 60;

    // Resize handler
    this.resizeObserver = new ResizeObserver(() => this.onResize());
    this.resizeObserver.observe(this.canvas.parentElement);
    this.onResize();
  }

  randomBlinkInterval() {
    return 2.5 + Math.random() * 4.5;
  }

  onResize() {
    const container = this.canvas.parentElement;
    const w = container.clientWidth;
    const h = container.clientHeight;
    this.renderer.setSize(w, h);
    this.camera.aspect = w / h;
    this.camera.updateProjectionMatrix();
  }

  loadTexture(emotion) {
    // В реальном проекте: загрузка текстуры из файла
    // const texture = new THREE.TextureLoader().load(`/avatar/textures/emotions/${emotion}.webp`);
    // this.avatarMesh.material.map = texture;
    // this.avatarMesh.material.needsUpdate = true;

    // Для демо: меняем цвет материала как индикатор
    const colors = {
      neutral: 0x4080ff,
      happy: 0x60a0ff,
      smile: 0x5090ff,
      excited: 0x80c0ff,
      sad: 0x3060a0,
      angry: 0xff4040,
      surprised: 0xff8040,
      thinking: 0x6080c0,
      confused: 0x8060a0,
      focused: 0x4060a0,
      speaking: 0x60a0ff,
      listening: 0x5080c0,
      blink: 0x204080
    };

    this.avatarMesh.material.color.setHex(colors[emotion] || 0x4080ff);
  }

  setEmotion(name, duration = 0.35) {
    if (!EMOTION_TEXTURES.hasOwnProperty(name)) return;
    this.currentEmotion = this.targetEmotion;
    this.targetEmotion = name;
    this.transitionDuration = duration;
    this.transitionProgress = 0;
    this.transitionStart = performance.now() / 1000;

    this.loadTexture(name);

    document.getElementById('emotion-label').textContent = name;
  }

  setSpeaking(active) {
    this.speaking = active;
  }

  setMouthLevel(value) {
    this.speechLevel = Math.max(0, Math.min(1, value));
  }

  blink() {
    this.blinkState.phase = 'closing';
    this.blinkState.duration = 0.09 + Math.random() * 0.09;
    this.blinkState.timer = 0;
  }

  update(delta) {
    const now = performance.now() / 1000;

    // Emotion transition
    if (this.transitionProgress < 1) {
      const elapsed = now - this.transitionStart;
      this.transitionProgress = Math.min(1, elapsed / this.transitionDuration);
    }

    // Blink system
    if (this.blinkEnabled) {
      this.updateBlink(delta, now);
    }

    // Eye movement
    if (this.eyeMovementEnabled) {
      this.updateEyeMovement(delta, now);
    }

    // Micro expressions
    if (this.microEnabled) {
      this.updateMicroExpressions(delta, now);
    }

    // Speech animation
    if (this.speaking) {
      const scale = 1 + this.speechLevel * 0.05;
      this.avatarMesh.scale.y = scale;
    } else {
      this.avatarMesh.scale.y = 1;
    }

    // Subtle breathing
    this.avatarMesh.position.y = Math.sin(now * 1.5) * 0.02;
    this.avatarMesh.rotation.y = Math.sin(now * 0.7) * 0.02;

    // FPS counter
    this.frameCount++;
    this.fpsTime += delta;
    if (this.fpsTime >= 0.5) {
      this.currentFPS = Math.round(this.frameCount / this.fpsTime);
      document.getElementById('fps-counter').textContent = `${this.currentFPS} FPS`;
      this.frameCount = 0;
      this.fpsTime = 0;
    }
  }

  updateBlink(delta, now) {
    const bs = this.blinkState;
    bs.timer += delta;

    if (bs.phase === 'open') {
      if (bs.timer >= bs.nextBlink) {
        bs.phase = 'closing';
        bs.duration = 0.09 + Math.random() * 0.09;
        bs.timer = 0;
      }
      bs.blinkOpacity = 1;
    } else if (bs.phase === 'closing') {
      const t = Math.min(1, bs.timer / (bs.duration * 0.4));
      const ease = easeInOutCubic(t);
      bs.blinkOpacity = 1 - ease * 0.3;
      if (t >= 1) {
        bs.phase = 'opening';
        bs.timer = 0;
      }
    } else if (bs.phase === 'opening') {
      const t = Math.min(1, bs.timer / (bs.duration * 0.6));
      const ease = easeInOutCubic(t);
      bs.blinkOpacity = 0.7 + ease * 0.3;
      if (t >= 1) {
        bs.phase = 'open';
        bs.timer = 0;
        bs.nextBlink = this.randomBlinkInterval();
      }
    }

    this.avatarMesh.material.opacity = bs.blinkOpacity;
  }

  updateEyeMovement(delta, now) {
    const es = this.eyeState;
    es.timer += delta;

    if (es.timer >= es.nextMove) {
      es.targetX = (Math.random() - 0.5) * 0.1;
      es.targetY = (Math.random() - 0.5) * 0.05;
      es.timer = 0;
      es.nextMove = 1.5 + Math.random() * 3;
    }

    const lerpSpeed = 3;
    es.currentX += (es.targetX - es.currentX) * lerpSpeed * delta;
    es.currentY += (es.targetY - es.currentY) * lerpSpeed * delta;

    this.avatarMesh.position.x = es.currentX;
  }

  updateMicroExpressions(delta, now) {
    const ms = this.microState;
    ms.timer += delta;

    if (ms.timer >= ms.nextMicro) {
      ms.intensity = 0.02 + Math.random() * 0.03;
      ms.timer = 0;
      ms.nextMicro = 3 + Math.random() * 5;
    }
  }

  render() {
    this.renderer.render(this.scene, this.camera);
  }

  dispose() {
    this.resizeObserver.disconnect();
    this.renderer.dispose();
  }
}

class MiniAvatarRenderer {
  constructor(canvas, emotion) {
    this.canvas = canvas;
    this.emotion = emotion;

    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 1.5));

    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(30, 1, 0.1, 100);
    this.camera.position.set(0, 0, 4);

    const ambientLight = new THREE.AmbientLight(0x4080ff, 0.6);
    this.scene.add(ambientLight);
    const keyLight = new THREE.DirectionalLight(0x80d0ff, 1.5);
    keyLight.position.set(2, 3, 4);
    this.scene.add(keyLight);

    const planeGeo = new THREE.PlaneGeometry(2.5, 3.2);
    const colors = {
      neutral: 0x4080ff, happy: 0x60a0ff, smile: 0x5090ff, excited: 0x80c0ff,
      sad: 0x3060a0, angry: 0xff4040, surprised: 0xff8040, thinking: 0x6080c0,
      confused: 0x8060a0, focused: 0x4060a0, speaking: 0x60a0ff, listening: 0x5080c0, blink: 0x204080
    };

    const planeMat = new THREE.MeshBasicMaterial({
      color: colors[emotion] || 0x4080ff,
      transparent: true,
      opacity: 1
    });

    this.mesh = new THREE.Mesh(planeGeo, planeMat);
    this.scene.add(this.mesh);

    this.resize();
  }

  resize() {
    const w = this.canvas.parentElement.clientWidth;
    const h = w;
    this.renderer.setSize(w, h);
    this.camera.aspect = 1;
    this.camera.updateProjectionMatrix();
  }

  render() {
    this.renderer.render(this.scene, this.camera);
  }

  dispose() {
    this.renderer.dispose();
  }
}

  let lastTime = performance.now();
  function animate() {
    requestAnimationFrame(animate);
    const now = performance.now();
    const delta = Math.min((now - lastTime) / 1000, 0.05);
    lastTime = now;

    mainAvatar.update(delta);
    mainAvatar.render();
  }
  animate();
}
